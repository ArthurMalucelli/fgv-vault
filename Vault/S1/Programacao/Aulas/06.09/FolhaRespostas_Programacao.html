R5 · INPUT + VALIDAÇÃO + FAIXAS

"Valide (int + categoria); erro → ERRO DE ENTRADA; senão calcule por faixa, adicional condicional"
x = input("Valor: ")
cat = input("Categoria: ").upper()
if not x.isdigit() or cat not in ["A","B","C"]:
    print("ERRO DE ENTRADA")
else:
    x = int(x)
    if x <= 10000:        # "até X" inclui X
        v = (x/1000) * 0.98
    elif x <= 50000:
        v = (x/1000) * 0.74
    elif x <= 100000:
        v = (x/1000) * 0.42
    else:
        v = 100            # fixo
    if cat in ["B","C"]:
        v *= 2             # dobro vale p/ fixo tb
    print(f"R$ {v:.2f}")
